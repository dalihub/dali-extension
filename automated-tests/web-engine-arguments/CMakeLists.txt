cmake_minimum_required(VERSION 3.16)
project(web-engine-arguments-tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(DALI_API_MAJOR 3 CACHE STRING "Installed DALi API major")
find_package(PkgConfig REQUIRED)
pkg_check_modules(DALI REQUIRED IMPORTED_TARGET dali${DALI_API_MAJOR}-adaptor)
enable_testing()
add_executable(arguments-test arguments-test.cpp)
target_link_libraries(arguments-test PRIVATE PkgConfig::DALI)
add_test(NAME web-engine-arguments COMMAND arguments-test)
