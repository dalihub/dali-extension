cmake_minimum_required(VERSION 3.16)
project(web-engine-cookie-tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Threads REQUIRED)
enable_testing()
set(ADAPTOR_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../dali-adaptor" CACHE PATH "DALi adaptor checkout")
set(source_root ${CMAKE_CURRENT_SOURCE_DIR}/../..)
foreach(backend ecore tcore)
  set(test_dir ${CMAKE_CURRENT_BINARY_DIR}/${backend})
  foreach(header ewk_cookie_manager.h ewk_cookie_manager_product.h wv_cookie_manager.h wv_cookie_manager_product.h dali/public-api/common/dali-namespace.h)
    file(WRITE ${test_dir}/${header} "#include \"native-mock.h\"\n")
  endforeach()
  add_executable(cookies-${backend} cookies-test.cpp
    ${source_root}/dali-extension/web-engine-chromium/${backend}/tizen-web-engine-cookie-manager.cpp)
  target_include_directories(cookies-${backend} PRIVATE ${test_dir} ${CMAKE_CURRENT_SOURCE_DIR}
    ${source_root} ${source_root}/dali-extension/web-engine-chromium/${backend} ${ADAPTOR_SOURCE})
  target_compile_definitions(cookies-${backend} PRIVATE DALI_NAMESPACE=Dali BACKEND_${backend})
  target_compile_options(cookies-${backend} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
  target_link_libraries(cookies-${backend} PRIVATE Threads::Threads)
  add_test(NAME cookies-${backend} COMMAND cookies-${backend})
endforeach()
